cmake_minimum_required(VERSION 3.20)
project(viergewinnt_dx12 LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "The DX12 bridge requires Windows or a Windows cross toolchain.")
endif()
find_package(JNI REQUIRED)
set(PT_FSR_SDK "" CACHE PATH "AMD FSR SDK 2.3.0 checkout root")
set(PT_XESS_SDK "" CACHE PATH "Intel XeSS 3 SDK checkout root")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/shaders/pathtrace.hlsl" PATHTRACE)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/shaders/present.hlsl" PRESENT)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/shaders/denoise.hlsl" DENOISE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS shaders/pathtrace.hlsl shaders/present.hlsl shaders/denoise.hlsl)
configure_file(shaders.h.in shaders.h @ONLY)
add_library(viergewinnt_dx12 SHARED dx12_bridge.cpp)
target_include_directories(viergewinnt_dx12 PRIVATE ${JNI_INCLUDE_DIRS} "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(viergewinnt_dx12 PRIVATE cxx_std_17)
target_link_libraries(viergewinnt_dx12 PRIVATE d3d12 dxgi d3dcompiler)
if(PT_FSR_SDK)
    if(NOT EXISTS "${PT_FSR_SDK}/Kits/FidelityFX/upscalers/include/ffx_upscale.h")
        message(FATAL_ERROR "PT_FSR_SDK must point to AMD FSR SDK 2.3.0")
    endif()
    target_compile_definitions(viergewinnt_dx12 PRIVATE PT_FSR)
    target_include_directories(viergewinnt_dx12 PRIVATE "${PT_FSR_SDK}/Kits/FidelityFX/api/include" "${PT_FSR_SDK}/Kits/FidelityFX/upscalers/include")
endif()
if(PT_XESS_SDK)
    if(NOT EXISTS "${PT_XESS_SDK}/inc/xess/xess_d3d12.h")
        message(FATAL_ERROR "PT_XESS_SDK must point to an Intel XeSS SDK checkout")
    endif()
    target_compile_definitions(viergewinnt_dx12 PRIVATE PT_XESS)
    target_include_directories(viergewinnt_dx12 PRIVATE "${PT_XESS_SDK}/inc")
endif()
if(MINGW)
    target_link_options(viergewinnt_dx12 PRIVATE -static-libgcc -static-libstdc++)
elseif(MSVC)
    target_compile_options(viergewinnt_dx12 PRIVATE /W4 /EHsc)
endif()
set_target_properties(viergewinnt_dx12 PROPERTIES PREFIX "" OUTPUT_NAME viergewinnt_dx12)
include(CTest)
if(BUILD_TESTING)
    add_executable(dx12_shader_check shader_check.cpp)
    target_include_directories(dx12_shader_check PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    target_link_libraries(dx12_shader_check PRIVATE d3dcompiler)
    add_test(NAME dx12_shaders COMMAND dx12_shader_check)
endif()
